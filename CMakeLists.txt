cmake_minimum_required(VERSION 3.20)
project(hexo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hexo SHARED src/hexo.cpp)
target_include_directories(hexo PUBLIC src)
if(MSVC)
  target_compile_options(hexo PRIVATE /O2 /W4)
else()
  target_compile_options(hexo PRIVATE -O3 -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(hexo PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
option(HEXO_NATIVE_CPU "Compile Native for this machine's x86 CPU (GCC only)" OFF)
set(HEXO_NATIVE_PGO "OFF" CACHE STRING "Native GCC profile mode: OFF, GENERATE, USE")
set_property(CACHE HEXO_NATIVE_PGO PROPERTY STRINGS OFF GENERATE USE)
set(HEXO_NATIVE_PROFILE_DIR "${CMAKE_BINARY_DIR}/native-profile" CACHE PATH "Native GCC profile directory")
string(TOUPPER "${HEXO_NATIVE_PGO}" _hexo_pgo)
if(NOT _hexo_pgo MATCHES "^(OFF|GENERATE|USE)$")
  message(FATAL_ERROR "HEXO_NATIVE_PGO must be OFF, GENERATE, or USE")
endif()
if(HEXO_NATIVE_CPU OR NOT _hexo_pgo STREQUAL "OFF")
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "Native CPU tuning and PGO currently require GCC")
  endif()
endif()
if(HEXO_NATIVE_CPU)
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i[3-6]86)$")
    message(FATAL_ERROR "HEXO_NATIVE_CPU currently supports x86 CPUs")
  endif()
  # MinGW GCC 15.2 can emit misaligned 256-bit stack stores with -march=native.
  target_compile_options(hexo PRIVATE -march=native -mprefer-vector-width=128)
endif()
if(NOT _hexo_pgo STREQUAL "OFF")
  get_filename_component(_hexo_profile_dir "${HEXO_NATIVE_PROFILE_DIR}" ABSOLUTE BASE_DIR "${CMAKE_BINARY_DIR}")
  set(_hexo_profile_prefix "${CMAKE_CURRENT_BINARY_DIR}")
  if(WIN32)
    # GCC compares this prefix literally against its backslash-separated cwd.
    string(REPLACE "/" "\\" _hexo_profile_prefix "${_hexo_profile_prefix}")
  endif()
  string(TOLOWER "${_hexo_pgo}" _hexo_profile_mode)
  set(_hexo_profile_flag "-fprofile-${_hexo_profile_mode}=${_hexo_profile_dir}")
  target_compile_options(hexo PRIVATE "${_hexo_profile_flag}" "-fprofile-prefix-path=${_hexo_profile_prefix}")
  target_link_options(hexo PRIVATE "${_hexo_profile_flag}")
  if(_hexo_pgo STREQUAL "USE")
    target_compile_options(hexo PRIVATE -Werror=missing-profile)
  endif()
endif()
set(HEXO_SEAL_SOURCE "" CACHE PATH "Optional external Ramora0/HexTicTacToe checkout")
if(HEXO_SEAL_SOURCE)
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${HEXO_SEAL_SOURCE}"
    OUTPUT_VARIABLE SEAL_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  file(WRITE "${CMAKE_BINARY_DIR}/seal_revision.txt" "${SEAL_REVISION}\n")
  add_library(hexo_seal SHARED tools/seal_adapter.cpp)
  target_include_directories(hexo_seal PRIVATE "${HEXO_SEAL_SOURCE}/cpp")
  if(NOT MSVC)
    target_compile_options(hexo_seal PRIVATE -O3)
    if(MINGW)
      target_link_options(hexo_seal PRIVATE -static-libgcc -static-libstdc++)
    endif()
  endif()
endif()
add_library(hexo_graph SHARED src/relational_graph.cpp)
if(MSVC)
  target_compile_options(hexo_graph PRIVATE /O2 /W4)
else()
  target_compile_options(hexo_graph PRIVATE -O3 -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(hexo_graph PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
add_library(hexo_gumbel SHARED src/gumbel_owner.cpp src/gumbel_batch.cpp src/gumbel_feed.cpp)
if(MSVC)
  target_compile_options(hexo_gumbel PRIVATE /O2 /W4)
else()
  target_compile_options(hexo_gumbel PRIVATE -O3 -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(hexo_gumbel PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
