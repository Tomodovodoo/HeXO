cmake_minimum_required(VERSION 3.20)
project(hexo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hexo SHARED src/hexo.cpp)
target_include_directories(hexo PUBLIC src)
if(MSVC)
  target_compile_options(hexo PRIVATE /O2 /W4)
else()
  target_compile_options(hexo PRIVATE -O3 -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(hexo PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
set(HEXO_SEAL_SOURCE "" CACHE PATH "Optional external Ramora0/HexTicTacToe checkout")
if(HEXO_SEAL_SOURCE)
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${HEXO_SEAL_SOURCE}"
    OUTPUT_VARIABLE SEAL_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  file(WRITE "${CMAKE_BINARY_DIR}/seal_revision.txt" "${SEAL_REVISION}\n")
  add_library(hexo_seal SHARED tools/seal_adapter.cpp)
  target_include_directories(hexo_seal PRIVATE "${HEXO_SEAL_SOURCE}/cpp")
  if(NOT MSVC)
    target_compile_options(hexo_seal PRIVATE -O3)
    if(MINGW)
      target_link_options(hexo_seal PRIVATE -static-libgcc -static-libstdc++)
    endif()
  endif()
endif()
add_library(hexo_graph SHARED src/relational_graph.cpp)
if(MSVC)
  target_compile_options(hexo_graph PRIVATE /O2 /W4)
else()
  target_compile_options(hexo_graph PRIVATE -O3 -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(hexo_graph PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
add_library(hexo_gumbel SHARED src/gumbel.cpp)
if(MSVC)
  target_compile_options(hexo_gumbel PRIVATE /O2 /W4)
else()
  target_compile_options(hexo_gumbel PRIVATE -O3 -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(hexo_gumbel PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
